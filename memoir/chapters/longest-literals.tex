\chapter{The Length of Longest Literals}
\section{Lower and upper bounds}

Denote $y_n := \EE[Y_n]$ for $n \ge 1$. Taking the expectation of both sides in the recurrence \eqref{eq:longest-literal-recurrence}, we have for $n \ge 1$,
\begin{align*}
  y_{n+2}
   & = \dfrac{u}{n}\sum\limits_{i=1}^n\left(y_{i} + y_{n+1-i}\right) + \dfrac{v}{n}\EE\left[\max\left(Y_{i}^{(1)}, Y_{n+1-i}^{(2)}\right)\right] \\
   & = \dfrac{2u}{n}\sum\limits_{i=1}^n y_{i} + \dfrac{v}{n}\EE\left[\max\left(Y_{i}^{(1)}, Y_{n+1-i}^{(2)}\right)\right].
\end{align*}

Using the inequality $\max(a,b) \le a + b$ for $a,b \ge 0$, we have
$$y_{n+2} \le \dfrac{2(u+v)}{n}\sum\limits_{i=1}^n y_{i}.$$
Since the right-hand side of the inequality is an increasing function of $(y_1, \ldots, y_n)$, the sequence $(y_n)$ is bounded above by the solution to the recurrence
\begin{equation}
  \label{eq:upper-bound}
  z_{n+2} = \dfrac{2(u+v)}{n}\sum\limits_{i=1}^n z_{i}, \quad n \ge 1, \quad z_1 = 1, z_2 = 0.
\end{equation}
Similarly, the inequality $\max(a,b) \ge \frac{1}{2}(a+b)$ gives a lower bound sequence
\begin{equation}
  \label{eq:lower-bound}
  z_{n+2} = \dfrac{2u+v}{n}\sum\limits_{i=1}^n z_{i}, \quad n \ge 1, \quad z_1 = 1, z_2 = 0.
\end{equation}
It is possible to employ a transfer theorem in analytic combinatorics to analyze the asymptotic behavior of these recurrences (Lemma \ref{lem:full-history-sum-asymptotics}). The lower bound can be improved by tightening the lower bound of the expectation of the maximum
$$\EE\left[\max\left(Y_{i}^{(1)}, Y_{n+1-i}^{(2)}\right)\right] \ge \max\left(\EE\left[Y_{i}^{(1)}\right], \EE\left[Y_{n+1-i}^{(2)}\right]\right) = \max\left(y_i, y_{n+1-i}\right).$$
Note that this bound is not sharp, the equality holds only when $Y_{i}^{(1)} = Y_{n+1-i}^{(2)}$ a.s. The lower bound recurrence is then given by
\begin{equation}
  \label{eq:lower-bound-tighter}
  z_{n+2} = \dfrac{2u}{n}\sum\limits_{i=1}^n z_{i} + \dfrac{v}{n}\sum\limits_{i=1}^n \max\left(z_i, z_{n+1-i}\right), \quad n \ge 1, \quad z_1 = 1, z_2 = 0.
\end{equation}
Detailed analysis of \eqref{eq:lower-bound-tighter} is given in the next section.

\section{Asymptotics of the lower bound recurrence}

% Process from n^{\alpha+o(1)} to \Theta(n^\alpha) to \sim Cn^\alpha.

The sequence defined by \eqref{eq:lower-bound-tighter} is bounded between those defined by \eqref{eq:upper-bound} and \eqref{eq:lower-bound}. If $2u+v > 1$, then both \eqref{eq:upper-bound} and \eqref{eq:lower-bound} grow as a pure power of $n$, without any lower-order factor like the logarithm. We conjecture that in this case \eqref{eq:lower-bound-tighter} also grows as a pure power of $n$. In this section, we always refer to $(z_n)$ as the sequence defined by \eqref{eq:lower-bound-tighter}, unless otherwise stated. The set of conditions on the parameters $u$ and $v$ is thus given by
\begin{equation}
  \label{eq:parameter-conditions}
  u,v \ge 0, \quad u < 1, \quad u+v\le1, \quad 2u+v > 1.
\end{equation}

In Theorem \ref{thm:eventually-increasing}, we will we to strengthen the conditions to
\begin{equation}
  \label{eq:parameter-conditions-stronger}
  u,v \ge 0, \quad \dfrac{1}{2} < u < 1, \quad u+v\le1.
\end{equation}

Let us carry out a heuristic to find out the power. Assume that $z_n \sim Cn^{\alpha}$ for some constants $c > 0$ and $\alpha > 0$. Then, substituting into \eqref{eq:lower-bound-tighter}, we get
\begin{align*}
  Cn^\alpha
   & \sim \dfrac{2u}{n}\sum\limits_{i=1}^n C i^\alpha + \dfrac{v}{n}\sum\limits_{i=1}^n C\max\left(i^\alpha, (n+1-i)^\alpha\right)                   \\
   & \sim C\left( \dfrac{2u}{n}\int_0^n C x^\alpha\,\mathrm{d}x + \dfrac{v}{n}\int_0^n C\max\left(x^\alpha, (n-x)^\alpha\right) \,\mathrm{d}x\right) \\
   & = C\left(\dfrac{2u}{n}\int_0^n x^\alpha\,\mathrm{d}x + \dfrac{2v}{n}\cdot \int_{n/2}^{n} x^\alpha \,\mathrm{d}x\right)                          \\
   & = C\left(\dfrac{2u}{n}\cdot \dfrac{n^{\alpha+1}}{\alpha+1} + \dfrac{2v}{n}\cdot \dfrac{n^{\alpha+1}-(n/2)^{\alpha+1}}{\alpha+1}\right)          \\
   & = \dfrac{C}{\alpha+1}\left(2(u+v) - v2^{-\alpha}\right)n^{\alpha}.
\end{align*}

Thus, we must have
\begin{equation}
  \label{eq:alpha-equation}
  \alpha + v2^{-\alpha} = 2(u+v) - 1.
\end{equation}

Note that $\alpha>0$, since otherwise $2u+v = 1$. A natural goal is to try to escape from the maximum operator. However, $(z_n)$ fluctuates in the first iterations, and the time where it begins to increase depends on the parameters $u$ and $v$. An extreme case is that when
$$u\to1/2+ \text{ and } v\to0+,$$
the sequence $(z_n)$ fluctuates for a long time before increasing consistently. It turns out that this property of \textit{eventual monotonicity} is sufficient to escape from the maximum operator. Indeed, assume that there exists $N_1$ such that $z_{n+1} \ge z_n$ for all $n \ge N_1$. Since $(z_n)$ is unbounded, there exists $N_2 \ge N_1$ such that for all $n \ge N_2$, we have both $z_{n+1} \ge \max\limits_{1\le i \le N_1} z_i$ and $z_{n+1} \ge z_n$. That is
$$\forall n \ge N_2, \quad z_{n+1} \ge \max_{1 \le i \le n} z_i.$$
Taking $N_3 = 2N_2-1$, we have
$$\forall n \ge N_3, \forall \dfrac{N_3+1}{2}\le i \le N_3, \quad z_{i} \ge z_{n+1-i}.$$

The question about when it stops fluctuating is itself interesting, which we leave open.
\begin{openquestion}
  \label{open:when-eventually-increasing}
  In terms of $u$ and $v$, what is the first index $N$ such that $(z_n)_{n\ge N}$ is increasing?
\end{openquestion}

We start with weaker results that do not require escaping from the maximum operator. Define
\begin{equation}
  F(x,y) = u(x+y) + v\max(x,y), x,y\in[0,1].
\end{equation}
We rewrite our recurrence as
\begin{equation}
  z_{n+2} = \dfrac{1}{n}\sum\limits_{i=1}^n F(z_i, z_{n+1-i}), \quad n \ge 1, \quad z_1 = 1, z_2 = 0.
\end{equation}

If $z_n\sim Cn^\alpha$, we can try to approximate $F$ as follows.

\begin{lemma}
  \label{thm:asymptoics-of-history}
  For every $\alpha \in (0,1]$, we have
  $$\dfrac{1}{n}\sum\limits_{i=1}^n F(i^\alpha, (n+1-i)^\alpha) \sim n^\alpha \int_0^1 F(x^\alpha, (1-x)^\alpha).$$
\end{lemma}

\begin{proof}
  Rewrite the asymptotics as a limit
  $$\dfrac{1}{n}\sum\limits_{i=1}^n F\left(\left(\dfrac{i}{n}\right)^\alpha, \left(1-\dfrac{i}{n}+\dfrac{1}{n}\right)^\alpha\right) \to \int_0^1 F(x^\alpha, (1-x)^\alpha).$$

  The function on the right-hand side is continuous on $[0,1]$, and the left-hand side is different from its Riemann sum only by a perturbation $1/n$ in the second argument. We show that the perturbation is negligible. Define for $x\in [0,1]$ the functions
  $$f_n(x) = F\left(x^\alpha, \left(1-x+\dfrac{1}{n}\right)^\alpha\right) \text{ and } f(x) = F(x^\alpha, (1-x)^\alpha),$$
  For $a,b,c \ge 0$ and $\alpha \in (0,1]$, we have
  $$|\max(a,b) - \max(a,c)| \le |b-c| \text{ and }
    (a+b)^\alpha \le a^\alpha + b^\alpha.$$
  Hence,
  \begin{align*}
    \|f_n-f\|_\infty
     & \le \sup\limits_{x\in[0,1]}(u+v)\left(\left(1-x+\dfrac{1}{n}\right)^\alpha - (1-x)^\alpha\right) \le (u+v)\dfrac{1}{n^\alpha} \to 0.
  \end{align*}

  Therefore,
  $$\left|\dfrac{1}{n}\sum\limits_{i=1}^n f_n\left(\dfrac{i}{n}\right) - \dfrac{1}{n}\sum\limits_{i=1}^n f\left(\dfrac{i}{n}\right)\right| \le \|f_n-f\|_\infty \to 0.$$
\end{proof}

Let
$$\Lambda(\alpha) = \int_0^1 F(x^\alpha, (1-x)^\alpha) \,\mathrm{d}x = \dfrac{2(u+v)-v2^{-\alpha}}{\alpha+1}.$$

Then $\Lambda$ is strictly decreasing. Indeed,
\begin{align*}
   & \Lambda'(\alpha) = \dfrac{v2^{-\alpha}\ln 2}{\alpha+1} - \dfrac{2(u+v)-v2^{-\alpha}}{(\alpha+1)^2} = \dfrac{v2^{-\alpha}((\alpha+1)\ln 2 + 1) - 2(u+v)}{(\alpha+1)^2}, \\
   & (2^{-\alpha}((\alpha+1)\ln 2 + 1))' = -2^{-\alpha}(\alpha+1)\ln^2(2) < 0.
\end{align*}
Hence,
$$(\alpha+1)^2\Lambda'(\alpha) < (\alpha+1)^2\Lambda'(0) = -(2u + (1-\ln 2)v) < 0.$$
Therefore, there exists a unique $\alpha$ such that $\Lambda(\alpha) = 1$, which is the solution to \eqref{eq:alpha-equation}. Furthermore, $\Lambda(\beta) < 1$ for all $\beta > \alpha$ and $\Lambda(\beta) > 1$ for all $\beta < \alpha$. That helps us prove a weaker asymptotic theorem.

\begin{theorem}
  \label{thm:weaker-asymptotics}
  Let $\alpha$ be the solution to \eqref{eq:alpha-equation}. Then, $z_n = n^{\alpha + o(1)}$.
\end{theorem}

\begin{proof}
  Let $A_n(\beta) = \dfrac{1}{n}\sum\limits_{i=1}^n F(i^\beta, (n+1-i)^\beta)$. Firstly, we prove the upper bound. Assume that $\beta > \alpha$. Then, $\Lambda(\beta) < 1$. Hence there exists $N$ such that for all $n \ge N$,
  $$A_n(\beta) < n^\beta < (n+2)^\beta.$$
  Let $M_n = \max_{1\le k \le n} \dfrac{z_i}{k^\beta}$. For $n \ge N$, we have
  $$z_{n+2} = \dfrac{1}{n}\sum_{i=1}^n F\left(z_i, z_{n+1-i}\right) \le M_n A_n(\beta) < M_n (n+2)^\beta.$$
  Hence, $\dfrac{z_{n+2}}{(n+2)^\beta} < M_n$. Or $(M_n)$ is eventually non-increasing. Therefore, $z_n = O(n^\beta)$.

  We argue similarly for the lower bound, with an adjustment for $z_2$. We have
  $$A_n(\beta) \sim n^\beta \sim (n+2)^\beta \text{ and } A_n(\beta) \sim \dfrac{1}{n}\sum\limits_{i=3}^{n-2} F(i^\beta, (n+1-i)^\beta).$$
  Assume that $\beta < \alpha$. Then, $\Lambda(\beta) > 1$. Hence there exists $N\ge 3$ such that for all $n \ge N$,
  $$\dfrac{1}{n}\sum\limits_{i=3}^{n-2} F(i^\beta, (n+1-i)^\beta) > (n+2)^\beta.$$
  Let $m_n = \min_{3\le i \le n-2} \dfrac{z_i}{i^\beta}$. For $n \ge N$, we have
  $$z_{n+2} \ge \dfrac{1}{n}\sum_{i=3}^{n-2} F\left(z_i, z_{n+1-i}\right) \ge m_n \dfrac{1}{n}\sum\limits_{i=3}^{n-2} F(i^\beta, (n+1-i)^\beta) > m_n (n+2)^\beta.$$
  Hence, $\dfrac{z_{n+2}}{(n+2)^\beta} > m_n$. Or $(m_n)$ is eventually increasing. Therefore, $z_n = \Omega(n^\beta)$.
\end{proof}

Next, we present the stronger big-$\Theta$ result. The proof is done by fine-grained induction. We provide integral bounds for the sum of the maximums to be used in the induction step.

\begin{lemma}
  \label{lem:integral-bounds}
  Let $f: [0, \infty) \to \mathbb{R}$ be a increasing function. For every integer $n \ge 1$, we have
  $$S_1 := \sum_{i=0}^{n-1} \max(f(i), f(n-1-i)) \le 2\int_{n/2}^n f(x) \, \mathrm{d}x \le S_2 :=\sum_{i=1}^{n} \max(f(i), f(n+1-i)). $$
\end{lemma}

\begin{proof}
  We first prove the upper bound on the first sum. We will use the following bounds for $t\ge 0$
  $$f(t) \le \int_{t}^{t+1} f(x) \, \mathrm{d}x \le 2 \int_{t+1/2}^{t+1} f(x) \, \mathrm{d}x.$$

  If $n=2k$, we have
  $$S_1 = 2 \sum_{i=k}^{2k-1} f(i) \le 2 \sum_{i=k}^{2k-1} \int_{i}^{i+1} f(x) \, \mathrm{d}x.$$
  If $n=2k+1$, we have
  $$ S_1 = 2 \sum_{i=k+1}^{2k} f(i) + f(k) \le 2 \sum_{i=k+1}^{2k} \int_i^{i+1} f(x) \, \mathrm{d}x + 2 \int_{k+1/2}^{k+1} f(x) \, \mathrm{d}x.$$

  The omit the proof for the lower bound on the second sum, while pointing out the bounds that would be used. For $t\ge 1$, we have
  $$f(t) \ge \int_{t-1}^{t} f(x) \, \mathrm{d}x \ge 2 \int_{t-1}^{t-1/2} f(x) \, \mathrm{d}x.$$
\end{proof}

\begin{theorem}
  \label{thm:theta-asymptotics}
  Let $\alpha$ be the solution to \eqref{eq:alpha-equation}. Then, $z_n = \Theta(n^{\alpha})$.
\end{theorem}
\begin{proof}
  We prove the lower bound. It is clear that $z_n > 0$ for all $n\ne 2$. The main idea is to isolate $z_2$. The induction hypothesis is $z_n \ge K (n+L)^\alpha$ for $n\neq 2$. The base cases and the exact constants are determined later. For $n \ge 3$, we have
  \begin{align*}
    nz_{n+2}
     & = 2u\sum\limits_{i=1}^n z_i + v\sum\limits_{i=1}^n \max(z_i, z_{n+1-i})                                                                \\
     & \ge K\left(2u\sum\limits_{i=1}^{n} (i+L)^\alpha + v\sum\limits_{i=1}^{n} \max((i+L)^\alpha, (n+1-i+L)^\alpha)\right) - K2u(2+L)^\alpha \\
     & \ge K\left(2u\int_0^{n} (x+L)^\alpha \, \mathrm{d}x + 2v\int_{n/2}^{n} (x+L)^\alpha \, \mathrm{d}x\right) - K2u(2+L)^\alpha.
  \end{align*}
  Let
  \begin{align*}
    A_n
     & = 2u\int_0^{n} (x+L)^\alpha \, \mathrm{d}x + 2v\int_{n/2}^{n} (x+L)^\alpha \, \mathrm{d}x             \\
     & = \dfrac{1}{\alpha+1}\left((2u+2v)(n+L)^{\alpha+1} - 2v\left(\dfrac{n}{2}+L\right)^{\alpha+1}\right).
  \end{align*}
  Using the expansions
  \begin{align*}
    (n+L)^{\alpha+1}                       & = n^{\alpha+1} + (\alpha+1)L n^\alpha + O(n^{\alpha-1}),                         \\
    \left(\dfrac{n}{2}+L\right)^{\alpha+1} & = 2^{-\alpha+1}n^{\alpha+1} + (\alpha+1)L2^{-\alpha} n^\alpha + O(n^{\alpha-1}),
  \end{align*}
  we get
  \begin{align*}
    A_n
     & = n^{\alpha+1} + L(2u+2v-2v2^{-\alpha})n^\alpha + O(n^{\alpha-1})        \\
     & = n^{\alpha+1} + L(\alpha + 1 - v2^{-\alpha})n^\alpha + O(n^{\alpha-1}).
  \end{align*}
  On the other hand,
  $$n(n+L+2)^\alpha = n^{\alpha+1} + \alpha(L+2) n^\alpha + O(n^{\alpha-1}).$$
  Hence,
  $$A_n - 2u(2+L)^\alpha - n(n+L+2)^\alpha = (L(1-v2^{-\alpha}) - 2\alpha) n^\alpha + O(n^{\alpha-1}).$$
  Since $1-v2^{-\alpha} > 0$, choose $L>\dfrac{2\alpha}{1-v2^{-\alpha}}$, then there exists $N$ such that for all $n \ge N$,
  $$A_n - 2u(2+L)^\alpha - n(n+L+2)^\alpha \ge 0,$$
  implying that $z_{n+2} \ge K(n+L+2)^\alpha$. Finally, we choose
  $K = \min\limits_{i\in\{1,3,\ldots, N+1\}}\dfrac{z_i}{(i+L)^\alpha} > 0$ to satisfy the base cases.

  Now we prove the upper bound. Our induction hypothesis is $z_n \le K (n-1)^\alpha$ for $n\ge 2$, for some constant $K > 1$. The base case $z_2=0$ is trivially satisfied. For $n\ge 2$, we have
  \begin{align*}
    nz_{n+2}
     & = 2u + 2u\sum\limits_{i=2}^n z_i + v\sum\limits_{i=1}^n \max(z_i, z_{n+1-i})                                      \\
     & \le K\left(2u\sum\limits_{i=1}^{n-1} i^\alpha + v\sum\limits_{i=0}^{n-1} \max(i^\alpha, (n-1-i)^\alpha)\right)+2u \\
     & \le K\left(2u\int_1^{n} x^\alpha \, \mathrm{d}x + 2v\int_{n/2}^{n} x^\alpha \, \mathrm{d}x\right) + 2u            \\
     & = K\left(2u\dfrac{n^{\alpha+1} - 1}{\alpha+1} + 2v\dfrac{n^{\alpha+1} - (n/2)^{\alpha+1}}{\alpha+1}\right)+2u     \\
     & = Kn^{\alpha+1}\dfrac{2u+2v - v2^{-\alpha}}{\alpha+1}- \dfrac{2Ku}{\alpha+1}+2u                                   \\
     & = Kn^{\alpha+1} - \dfrac{2Ku}{\alpha+1} + 2u.
  \end{align*}
  The first inequality is valid by noticing that
  $$\max(z_1, z_n) = \max(1, z_n) \le \max(1, K(n-1)^\alpha) = \max(0, K(n-1)^\alpha).$$
  The second inequality uses Lemma \ref{lem:integral-bounds}. Choose $K=\alpha+1 > 1$, then $z_{n+2} \le Kn^\alpha \le K(n+2)^\alpha$.
\end{proof}

  In the proof of the lower bound we exploit the fact that $(n+L)^{\alpha+1}$ dominates $n(n+L+2)^{\alpha+1}$.

We proceed to the exact asymptotics $z_n \sim Cn^\alpha$. The idea is to use loose inequalities to prove easier quantitative properties, then upgrade step by step with tighter inequalities.

\begin{lemma}[Vanishing difference]
  \label{lem:vanishing-difference}
  We have $|z_{n+1} - z_n| \to 0$.
\end{lemma}

\begin{proof}
  For $n \ge 2$, we have
  \begin{align*}
    nz_{n+2}     & = 2u\sum_{i=1}^n z_i + v\sum_{i=1}^n \max(z_i, z_{n+1-i}),       \\
    (n-1)z_{n+1} & = 2u\sum_{i=1}^{n-1} z_i + v\sum_{i=1}^{n-1} \max(z_i, z_{n-i}).
  \end{align*}
  Hence,
  \begin{equation}
    \label{eq:difference}
    \begin{aligned}
      n(z_{n+2} - z_{n+1})
       & = 2uz_n + v\max(z_n,z_1) - z_{n+1} + v\sum_{i=1}^{n-1} (\max(z_i, z_{n+1-i}) - \max(z_i, z_{n-i}))      \\
       & = 2uz_n + v\max(z_n,z_1) - z_{n+1} + v\sum_{i=1}^{n-1} (\max(z_{n-i}, z_{i+1}) - \max(z_{n-i}, z_{i})).
    \end{aligned}
  \end{equation}
  Using $\max(z_n, z_1) \le z_n+z_1$, we have
  \begin{align*}
    n(z_{n+2} - z_{n+1})
     & \le 2uz_n + v(z_n + z_1) - vz_{n+1} + v\sum_{i=1}^{n-1} (\max(z_{n-i}, z_{i+1}) - \max(z_{n-i}, z_{i})) \\
     & \le 2uz_n + vz_1 + v(z_n - z_{n+1}) + v\sum_{i=1}^{n-1} (\max(z_{n-i}, z_{i+1}) - \max(z_{n-i}, z_{i}))
  \end{align*}
  Using $|\max(a,b) - \max(a,c)| \le |b-c|$, we have
  \begin{align*}
    n|z_{n+2} - z_{n+1}|
     & \le 2uz_n + vz_1 + v|z_{n+1} - z_n| + v\sum_{i=1}^{n-1} |z_{i+1} - z_{i}| = 2uz_n + vz_1 + v\sum_{i=1}^{n}|z_{i+1} - z_i| \\
     & \le 2uz_n + vz_1 + nv\sup_{1\le i \le n}|z_{i+1} - z_i|.
  \end{align*}
  Since $u < 1$, we have $\alpha < 1$. By $z_n = \Theta(n^\alpha)$, we have $\dfrac{1}{n}(2uz_n + vz_1) \to 0$, or
  $$|z_{n+2} - z_{n+1}| \le v\sup_{1\le i \le n}|z_{i+1} - z_i| + o(1).$$
  We have $v < 1$ since otherwise, $2u+v > 1$ would be violated. Hence $(|z_{i+1} - z_i|)$ is bounded. Let $M = \sup_{n \ge 1} |z_{n+1} - z_n|$. Let $\epsilon > 0$, then there exists $N$ such that for all $n \ge N$, $|z_{n+1} - z_n| < M + \epsilon$. Therefore,
  $$|z_{n+2} - z_{n+1}| \le v(M+\epsilon) + o(1).$$
  Let $n\to\infty$, we have $M \le v(M+\epsilon)$. Since $\epsilon > 0$ is arbitrary, we have $M = 0$.
\end{proof}

Since $(z_n)$ is unbounded, we have the following corollary.

\begin{corollary}[Eventual exceedance]
  \label{cor:eventual-exceed}
  Let $\alpha$ be the solution to \eqref{eq:alpha-equation}. Then, for every $M > 0$, there exists $N$ such that for all $n \ge N$, $z_n > M$.
\end{corollary}

\begin{lemma}[Increment identity]
  \label{lem:increment-identity}
  For $n\ge 1$, let $\Delta_n=z_{n+1}-z_n$. Then for all sufficiently large $n$, there exist numbers $\theta_{n,i}\in [0,1]$ for $1\le i\le n-1$ such that
  \begin{equation}
    \label{eq:increment-identity}
    n\Delta_{n+1} = (2u+v-1)z_n-\Delta_n +v\sum_{i=1}^{n-1}\theta_{n,i}\Delta_i.
  \end{equation}
\end{lemma}

\begin{proof}
  We start from \eqref{eq:difference}
  $$n\Delta_{n+1} = 2uz_n + v\max(z_n,z_1) - z_{n+1} + v\sum_{i=1}^{n-1} (\max(z_i, z_{n+1-i}) - \max(z_i, z_{n-i})).$$
  Now with Corollary \ref{cor:eventual-exceed}, we have $\max(z_n,z_1)=z_n$ for all sufficiently large $n$. Hence, for all sufficiently large $n$,
  $$n\Delta_{n+1} = (2u+v-1)z_n -\Delta_n + v\sum_{i=1}^{n-1} (\max(z_{n-i}, z_{i+1}) - \max(z_{n-i}, z_{i})).$$
  For every $a \ge 0$, the function $x\mapsto \max(a,x)$ is increasing and $1$-Lipschitz. Hence for each $n,i$ there is some $\theta_{n,i}\in[0,1]$ satisfying \eqref{eq:increment-identity}.
\end{proof}

To this step, we have isolated the increment sequence $(\Delta_n)$ from the original sequence $(z_n)$. Our next goal is to prove that $(\Delta_n)$ is \textit{eventually positive}. That would be done if in \eqref{eq:increment-identity}, we can absorb a lower bound of $-\Delta_n + v\sum_{i=1}^{n-1}\theta_{n,i}\Delta_i$ into $(2u+v-1)z_n$. We already have $|\Delta_n| = o(z_n)$ since $(\Delta_n)$ is bounded while $z_n\to\infty$. Because we do not know much about $\theta_{n,i}$, we can only apply the worst case to them. A basic inequality for absolute value leads to showing that $\sum_{i=1}^{n}|\Delta_i| = o(z_n)$. Unfortunately, if it is true that $z_n\sim Cn^\alpha$ and $z_n$ is eventually increasing, then $\sum_{i=1}^{n}|\Delta_i| \sim Cn^\alpha$. Therefore, we need a tighter lower bound. We will use the \textit{negative variation} of $\Delta_n$. For every $x\in\mathbb{R}$, we have
\begin{equation}
  x_{-} := \max(-x,0) = \dfrac{1}{2}(|x| - x) \ge -x \iff x \ge -x_{-}.
\end{equation}
Define the negative variation series $(\Delta_n)$ as
\begin{equation}
  \label{eq:negative-variation}
  V_n := \sum_{i=1}^{n} (\Delta_i)_{-}.
\end{equation}
Then $(V_n)$ is increasing. We state the criterion for eventual increase of $(z_n)$ in terms of $(V_n)$ as follows.

\begin{lemma}[Negative variation criterion]
  If $V_n = o(z_n)$, then $(z_n)$ is eventually increasing.
\end{lemma}

\begin{proof}
  By Lemma~\ref{lem:increment-identity}, for all sufficiently large $n$,
  $$n\Delta_{n+1} \ge (2u+v-1)z_n-|\Delta_n|-vV_n.$$
  Since $\Delta_n$ is bounded and $z_n\to\infty$, we have $|\Delta_n|=o(z_n)$. By assumption, $V_n = o(z_n)$. Hence for all sufficiently large $n$,
  $$n\Delta_{n+1} \ge \dfrac{1}{2}(2u+v-1)z_n > 0.$$
\end{proof}

It would be perfect if we could prove that $(V_n)$ is bounded, as this is indeed the truth. However, it is already good to prove that $V_n = o(n^\alpha)$, since we already have $z_n = \Theta(n^\alpha)$.

\begin{lemma}
  We have $V_n = O(n^v)$.
\end{lemma}

\begin{proof}
  Again, we start from \eqref{eq:increment-identity}. There exists $N$ such that for all $n\ge N$,
  $$n\Delta_{n+1} = (2u+v-1)z_n - \Delta_n + v\sum_{i=1}^{n-1}\theta_{n,j}\Delta_i.$$
  Let $M = \sup |\Delta_n|$. If $\Delta_{n+1}<0$, then
  \begin{align*}
    n(\Delta_{n+1})_-
     & = -n\Delta_{n+1} = |\Delta_n| - (2u+v-1)z_n - v\sum_{j=1}^{n-1}(\Delta_j)_- \\
     & \le  vV_{n-1} + |\Delta_n| - (2u+v-1)z_n                                    \\
     & \le vV_{n-1} + M.
  \end{align*}
  Hence, $(\Delta_{n+1})_- \le \dfrac{v}{n}V_{n-1}+\dfrac{M}{n} \le \dfrac{v}{n}V_{n} + \dfrac{M}{n}$. Therefore,
  \begin{equation}
    \label{eq:negative-variation-bound}
    V_{n+1} \le \left(\dfrac{v}{n}+1\right)V_{n}+\dfrac{M}{n}.
  \end{equation}
  Note that if $\Delta_{n+1} \ge 0$, then $(\Delta_{n+1})_- = 0$, leading to $V_{n+1} = V_n$, or \eqref{eq:negative-variation-bound} holds automatically. We then have
  $$V_{n+1} + \dfrac{M}{v} \le \left(\dfrac{v}{n}+1\right)\left(V_{n} + \dfrac{M}{v}\right) \le \prod_{i=N}^n\left(\dfrac{v}{i}+1\right)\left(V_N + \dfrac{M}{v}\right).$$
  We are left to estimate the product. Taking the logarithm gives
  \begin{align*}
    \ln\left(\prod_{i=N}^n\left(1+\frac{v}{i}\right) \right)
     & = \sum_{i=N}^n\ln\left(1+\frac{v}{i}\right)     \\
     & \le \sum_{i=N}^n \frac{v}{i} = v(H_n - H_{N-1}) \\
     & \le vH_n \le v\ln n + C,
  \end{align*}
  where $H_n=\sum_{i=1}^n \frac{1}{i} = \ln n + \gamma + O(1/n)$ is the $n$-th harmonic number and $C$ is some constant. Taking back the exponential of both sides yields
  $$\prod_{i=N}^n \left( 1 + \frac{v}{i} \right) \le e^{C} n^v = O(n^v).$$
\end{proof}

We state our checkpoint as a theorem.

\begin{theorem}
  \label{thm:eventually-increasing}
  If $v < \alpha$, then $(z_n)$ is eventually increasing.
\end{theorem}

Let us reduce the condition $v < \alpha$ in terms of $u$ and $v$. Define
$$f(\alpha) = \alpha + 1 - 2(u + v) + v 2^{-\alpha}, \alpha\in[0,1].$$
Then $f'(\alpha) = 1 - v2^{-\alpha}\ln2 > 1-\ln2 > 0$ for all $\alpha > 0$. Hence $v < \alpha$ is equivalent to $f(v) < 0$ i.e.,
$$v + 1 - 2(u + v) + v 2^{-v} < 0 \iff u > \frac{1}{2}(1 - v + v 2^{-v}).$$
If we impose conditions \ref{eq:parameter-conditions-stronger}, then the above inequality is satisfied. The case $v\ge \alpha$ is left open. With $(z_n)$ being eventually increasing, it has a nicer recurrence without the whole history.

\begin{lemma}
  There exists $N\ge 1$ such that the sequence $(z_n)_{n\ge N}$ defined by \ref{eq:lower-bound-tighter} satisfies the recurrence
  \begin{equation}
    \label{eq:delayed-recurrence}
    nz_{n+2}-(n-1)z_{n+1} = 2(u+v)z_{n} - vz_{\lceil (n-1)/2\rceil}.
  \end{equation}
  with $z_{N+1} \ge z_{N} \ge \ldots \ge z_{\lceil (N-1)/2\rceil} \ge 1$.
\end{lemma}

\begin{proof}
  Since $(z_n)$ is eventually increasing, there exists $N$ such that for all $n\ge N$, we have $z_n \ge z_i$ for all $1\le i \le n-1$ and $z_n \ge 1$. Hence, for all $n\ge N+1$,
  \begin{align*}
    nz_{n+2}     & = 2u\sum_{i=1}^n z_i + 2v\sum_{i=\lceil n/2\rceil+1}^n z_i + \delta_nvz_{\lceil n/2\rceil}                      \\
    (n-1)z_{n+1} & = 2u\sum_{i=1}^{n-1} z_i + 2v\sum_{i=\lceil (n-1)/2\rceil+1}^{n-1} z_i + \delta_{n-1}vz_{\lceil (n-1)/2\rceil},
  \end{align*}
  where $\delta_n = 1$ if $n$ is odd and $0$ if $n$ is even. If $n=2k$, then $\lceil n/2\rceil = \lceil (n-1)/2\rceil = k$, $\delta_n = 0$ and $\delta_{n-1} = 1$. Hence,
  $$2kz_{2k+2} - (2k-1)z_{2k+1} = 2(u+v)z_{2k} - vz_{k}.$$
  If $n=2k+1$, then $\lceil n/2\rceil > \lceil (n-1)/2\rceil = k$, $\delta_n = 1$ and $\delta_{n-1} = 0$. Hence,
  $$(2k+1)z_{2k+3} - (2k)z_{2k+2} = 2(u+v)z_{2k+1} - 2vz_{k+1} + vz_{k+1} = (2k+1)(u+v)z_{2k+1} - vz_{k+1}.$$
  In both cases, we arrive at \eqref{eq:delayed-recurrence}
\end{proof}

\begin{theorem}
  \label{thm:exact-asymptotics}
  Let $(z_n)$ be the defined by \eqref{eq:delayed-recurrence} and $\alpha$ be the solution to \eqref{eq:alpha-equation}. Then $z_n\sim C n^\alpha$ for some constant $C > 0$.
\end{theorem}

\begin{proof}
  Let $x_n=\dfrac{z_n}{(n-2)^\alpha}$ for $n\ge N$. We will prove that $(x_n)$ converges by proving that $\sum_{n\ge 0}|x_{n+1}-x_n|$ converges. We have, for $n\ge 5$,
  \begin{align*}
         & n^{\alpha+1}x_{n+1} - (n-1)^{\alpha+1}x_n = 2(u+v)(n-2)^{\alpha}x_{n-1} - v\left(\left\lfloor \dfrac{n-1}{2}\right\rfloor-1\right)^{\alpha}x_{\lfloor (n-1)/2\rfloor}                                                 \\
    \iff & x_{n+1} = \left(1-\dfrac{1}{n}\right)^{\alpha+1}x_n + \dfrac{2(u+v)}{n}\left(1-\dfrac{2}{n}\right)^{\alpha}x_{n-1} - \dfrac{v}{n}\left(\dfrac{\lfloor (n-1)/2\rfloor-1}{n}\right)^{\alpha}x_{\lfloor (n-1)/2\rfloor}.
  \end{align*}
  We have
  \begin{align*}
     & \left(1-\dfrac{1}{n}\right)^{\alpha+1} = 1 - \dfrac{\alpha+1}{n} + O\left(\dfrac{1}{n^2}\right),                                                                                                  \\
     & \dfrac{1}{n}\left(1-\dfrac{2}{n}\right)^{\alpha} = \dfrac{1}{n} + O\left(\dfrac{1}{n^2}\right),                                                                                                   \\
     & -\frac{1}{n}\left(\dfrac{\lfloor (n-1)/2\rfloor-1}{n}\right)^{\alpha}=-\frac{1}{n} \left( 2^{-\alpha} + O\left(\frac{1}{n}\right) \right) = -\frac{2^{-\alpha}}{n} + O\left(\frac{1}{n^2}\right).
  \end{align*}
  By the previous lemma, $(z_n)$ is bounded. Therefore,
  \begin{align*}
    z_{n+1}-z_n
     & = -\dfrac{\alpha+1}{n}z_n + \dfrac{2(u+v)}{n}z_{n-1} - \frac{v2^{-\alpha}}{n}z_{\lfloor (n-1)/2\rfloor} + O\left(\frac{1}{n^2}\right) \\
     & = -\dfrac{2(u+v)}{n}(z_n-z_{n-1}) + \frac{v2^{-\alpha}}{n}(z_{n}-z_{\lfloor (n-1)/2\rfloor}) + O\left(\frac{1}{n^2}\right).
  \end{align*}

  Let $\nabla z_n = z_{n} - z_{n-1}$. Then, there exists a constant $L>0$ such that for every $n\ge 5$,
  $$|\nabla z_{n+1}| \le \dfrac{2(u+v)}{n}|\nabla z_{n}| + \frac{v2^{-\alpha}}{n}\sum_{i=\lfloor (n+1)/2\rfloor}^{n}|\nabla z_{i}| + \dfrac{L}{(n+1)^2}.$$
  Equivalently, for every $n\ge 6$,
  $$|\nabla z_n| \le \dfrac{2(u+v)}{n-1}|\nabla z_{n-1}| + \frac{v2^{-\alpha}}{n-1}\sum_{i=\lfloor n/2\rfloor}^{n-1}|\nabla z_{i}| + \dfrac{L}{n^2}.$$

  Let $u_m = \max_{2^m \le n < 2^{m+1}} |\nabla z_n|$. Then, for every $n\in [2^m, 2^{m+1})$, we have
  \begin{align*}
     & \dfrac{1}{n-1}|\nabla z_{n-1}| \le \dfrac{1}{n-1}\max_{2^m-1\le i < 2^{m+1}-2}|\nabla z_i| = \dfrac{1}{n-1}\max(u_{m-1},u_m),                                                                      \\
     & \dfrac{1}{n-1}\sum_{i=\lfloor n/2\rfloor}^{n-1}|\nabla z_{i}| \le \dfrac{n+1}{2(n-1)}\max_{2^{m-1}\le i < 2^{m+1}} |\nabla z_i| \le \left(\dfrac{1}{2}+\dfrac{1}{n-1}\right) \max(u_{m-1}, u_{m}).
  \end{align*}
  Therefore,
  \begin{align*}
    |\nabla z_{n}|
     & \le \dfrac{2(u+v)}{n-1}\max(u_{m-1},u_m) + v2^{-\alpha}\left(\dfrac{1}{2}+\dfrac{1}{n-1}\right)\max(u_{m-1}, u_m) + \dfrac{L}{2^{2m}} \\
     & = \left(\dfrac{\alpha+1}{n-1}+ v2^{-\alpha-1}\right)\max(u_{m-1}, u_m) + \dfrac{L}{2^{2m}}.
  \end{align*}
  Since $v2^{-\alpha} = 2(u+v)-\alpha-1 < 1$, we can choose $m_0$ large enough such that for every $m\ge m_0$ and $n\in [2^m, 2^{m+1})$, there exists $\gamma_1$ such that $\dfrac{\alpha+1}{n-1}+ v2^{-\alpha-1} < \gamma_1 < \dfrac{1}{2}$. Taking the maximum over $n\in [2^m, 2^{m+1})$, we have
  $$u_m \le \gamma_1\max(u_{m-1}, u_m) + \dfrac{L}{2^{2m}}.$$

  If $u_{m-1} > u_m$, then $u_m \le \gamma_1 u_{m-1} + \dfrac{L}{2^{2m}}.$

  If $u_{m-1} \le u_m$, then $u_m \le \dfrac{L}{2^{2m}(1-\gamma_1)}.$

  In both cases, we have
  \begin{align*}
    u_m
     & \le \gamma_1 u_{m-1} + \dfrac{L}{2^{2m}(1-\gamma_1)}                                                             \\
     & \le \sum_{i=m_0+1}^m \gamma_1^{m-i}\dfrac{L}{2^{2i}(1-\gamma_1)} + \gamma_1^{m-m_0}u_{m_0}                       \\
     & < \dfrac{L}{(1-\gamma_1)}\sum_{i=m_0+1}^m \gamma_1^{m-i}\left(\dfrac{1}{4}\right)^{i} + \gamma_1^{m-m_0}u_{m_0}.
  \end{align*}
  Choose $\gamma_2$ such that $\max\left(\gamma_1, \dfrac{1}{4}\right) < \gamma_2 < \dfrac{1}{2}$. We have
  \begin{align*}
    u_m < \dfrac{L}{(1-\gamma_1)}\sum_{i=m_0+1}^m \gamma_2^m + \gamma_2^{m-m_0}u_{m_0} \le L'm\gamma_2^m,
  \end{align*}
  for some constant $L'>0$. Choose $\gamma_3$ such that $\gamma_2 < \gamma_3 < \dfrac{1}{2}$. Then there exists $m_1>m_0$ such that for every $m\ge m_1$, we have $L'm\gamma_2^m < L'\gamma_3^m$, implying that,
  $$u_m < L'\gamma_3^m.$$
  Therefore,
  \begin{align*}
    \sum_{n=0}^\infty|\nabla z_{n}| = \sum_{m=0}^\infty \sum_{n=2^m}^{2^{m+1}-1}|\nabla z_n| = \sum_{m=0}^\infty 2^m u_m < \sum_{m=0}^{m_1-1} 2^m u_m  + L'\sum_{m=m_1}^\infty (2\gamma_3)^{m} < \infty.
  \end{align*}
  Hence, $(z_n)$ converges to some constant $C>0$. Thus, $x_n \sim Cn^\alpha$.
\end{proof}



\section{Asymptotics}

With the results that the lower and upper bounds grow as big-$\Theta$ of a pure power of $n$, it is plausible to conjecture the same situation for the original sequence $(y_n)$. We adapt the contraction method (e.g, \cite{neininger2004general}). The original idea is as follows.
\begin{enumerate}
  \item Guess the scaling factor, then derive a normalized recurrence.
  \item Derive a limiting fixed-point equation from the normalized recurrence.
  \item Prove that the limiting equation has a unique solution in a suitable metric space using a contraction argument.
  \item Prove that the normalized sequence converges in distribution to that unique solution.
\end{enumerate}
We discuss the reality that the contraction method is \textit{not} directly applicable to our case. Let $X_n = \dfrac{Y_n}{n^\alpha}$ for some $\alpha > 0$ to be determined. We have the recurrence for $X_n$

\begin{equation}
  \label{eq:scaled-recurrence}
  \begin{aligned}
    X_{n+2}
     & \stackrel{d}{=} I\left(\left(\dfrac{U_n}{n+2}\right)^\alpha X_{U_n}^{(1)} + \left(\dfrac{n+1-U_n}{n+2}\right)^\alpha X_{n+1-U_n}^{(2)}\right)   \\
     & \myspace{50} + J\max\left(\left(\dfrac{U_n}{n+2}\right)^\alpha X_{U_n}^{(1)}, \left(\dfrac{n+1-U_n}{n+2}\right)^\alpha X_{n+1-U_n}^{(2)}\right)
  \end{aligned}
\end{equation}

The idea is to derive a limiting equation, then prove that the scaled sequence $(X_n)$ converges in distribution to a unique solution of the limiting equation. In our case, we state the limiting equation as follows.

\begin{lemma}
  \label{lem:limiting-equation}
  Suppose that $X_n \stackrel{d}{\to} X$. Then, $X$ satisfies the following distributional equation
  \begin{equation}
    \label{eq:limiting-equation}
    X \stackrel{d}{=} I\left(U^\alpha X^{(1)} + (1-U)^\alpha X^{(2)}\right) + J\max(U^\alpha X^{(1)}, (1-U)^\alpha X^{(2)}),
  \end{equation}
  where $U \sim \mathrm{Unif}[0,1]$.
\end{lemma}

The proof of Lemma \ref{lem:limiting-equation} is given in Appendix \ref{subsec:proof-limiting-equation}. Our first goal is to prove that there exists a unique solution $(X,\alpha)$ to \eqref{eq:limiting-equation}. We define the operator induced by the right-hand side of \eqref{eq:limiting-equation} by $T: \D \to \D$ such that
\begin{equation}
  T_{\alpha}(\mu) = \mathcal{L}\left(I\left(U^\alpha X^{(1)} + (1-U)^\alpha X^{(2)}\right) + J\max(U^\alpha X^{(1)}, (1-U)^\alpha X^{(2)})\right), \quad X^{(1)}, X^{(2)} \stackrel{i.i.d}{\sim}\mu.
\end{equation}
Observe that $\delta_0$ is always a solution to \eqref{eq:limiting-equation}, which is undesired. Hence, we have to choose a space that does not contain $\delta_0$. The operator is not a contraction in the Wasserstein metric. The Zolotarev metric \cite{zolotarev1977approximation} gives tighter bounds for proving contraction once the mean is fixed, but $T_\alpha$ does not preserve the mean.
